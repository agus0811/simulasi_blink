// Pin 13 terhubung ke LED internal & eksternal
const int ledPin = 13;

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  digitalWrite(ledPin, HIGH); // LED Menyala
  delay(1000);
  digitalWrite(ledPin, LOW);  // LED Mati
  delay(1000);
}
